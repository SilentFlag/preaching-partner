CREATE TABLE tokens(refresh BOOL, token VARCHAR(255));

CREATE TABLE congregation(id INTEGER PRIMARY KEY, name VARCHAR(255) NOT NULL, updated INTEGER);

CREATE TABLE categories(id INTEGER PRIMARY KEY, name VARCHAR(255) NOT NULL, prefix VARCHAR(255) NOT NULL, congregation INTEGER, updated INTEGER, CONSTRAINT fk_congregation FOREIGN KEY (congregation) REFERENCES congregation(id));

CREATE TABLE users(id INTEGER PRIMARY KEY, name VARCHAR(255) NOT NULL);

CREATE TABLE service_group(id INTEGER PRIMARY KEY, name VARCHAR(255) NOT NULL, congregation INTEGER, elder INTEGER, updated INTEGER, CONSTRAINT fk_congregation FOREIGN KEY (congregation) REFERENCES congregation(id), CONSTRAINT fk_elder FOREIGN KEY (elder) REFERENCES users(id));

CREATE TABLE maps(id INTEGER PRIMARY KEY, assignee INT, assigner INT, category INT NOT NULL, file_name VARCHAR(255), CONSTRAINT fk_category FOREIGN KEY (category) REFERENCES categories(id));

CREATE TABLE streets(id INTEGER PRIMARY KEY, map_id, name VARCHAR(255), CONSTRAINT fk_map_id FOREIGN KEY (map_id) REFERENCES maps(id));

CREATE TABLE addresses(id INTEGER PRIMARY KEY, street_id INTEGER, number VARCHAR(255), tags BLOB, visited BOOL, CONSTRAINT fk_street_id FOREIGN KEY (street_id) REFERENCES streets(id));

CREATE TABLE logs(code INTEGER, message TEXT, timestamp INTEGER);
