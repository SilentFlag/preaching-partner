CREATE TABLE tokens(refresh BOOL, token VARCHAR(255));

CREATE TABLE congregation(id INTEGER, name VARCHAR(255) NOT NULL, updated INTEGER);

CREATE TABLE categories(id INTEGER, name VARCHAR(255) NOT NULL, prefix VARCHAR(255) NOT NULL, congregation INTEGER, updated INTEGER, CONSTRAINT fk_congregation FOREIGN KEY (congregation) REFERENCES congregation(id));

CREATE TABLE users(id INTEGER, firstname VARCHAR(255) NOT NULL, lastname VARCHAR(255) NOT NULL, congregation INTEGER, service_group INTEGER, CONSTRAINT fk_congregation FOREIGN KEY (congregation) REFERENCES congregation(id), CONSTRAINT fk_service_group FOREIGN KEY (service_group) REFERENCES service_group(id));

CREATE TABLE service_group(id INTEGER, name VARCHAR(255) NOT NULL, congregation INTEGER, elder INTEGER, updated INTEGER, CONSTRAINT fk_congregation FOREIGN KEY (congregation) REFERENCES congregation(id), CONSTRAINT fk_elder FOREIGN KEY (elder) REFERENCES users(id));

CREATE TABLE maps(id INTEGER, assignee INT, assigner INT, category INT NOT NULL, file_name VARCHAR(255), updated INTEGER, CONSTRAINT fk_category FOREIGN KEY (category) REFERENCES categories(id));
